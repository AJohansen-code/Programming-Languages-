// Full ShiCode program combining functions, conditionals, loops, and calls
fun factorial(n) {
    let result = 1;
    while (n > 1) {
        if (n % 2 == 0) {
            result = result * n;
        } else {
            result = result * n;
        }
        n = n - 1;
    }
    return result;
}

print factorial(5);
